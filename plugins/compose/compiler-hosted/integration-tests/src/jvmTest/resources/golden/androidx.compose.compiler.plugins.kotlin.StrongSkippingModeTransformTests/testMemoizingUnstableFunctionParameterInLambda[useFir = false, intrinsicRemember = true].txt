@Composable
fun Test(foo: Foo, bar: Bar, %composer: Composer?, %changed: Int) {
  %composer = %composer.startRestartGroup(<>)
  val %dirty = %changed
  if (%changed and 0b0110 == 0) {
    %dirty = %dirty or if (if (%changed and 0b1000 == 0) {
      %composer.changed(foo)
    } else {
      %composer.changedInstance(foo)
    }
    ) 0b0100 else 0b0010
  }
  if (%changed and 0b00110000 == 0) {
    %dirty = %dirty or if (if (%changed and 0b01000000 == 0) {
      %composer.changed(bar)
    } else {
      %composer.changedInstance(bar)
    }
    ) 0b00100000 else 0b00010000
  }
  if (%composer.shouldExecute(%dirty and 0b00010011 != 0b00010010, %dirty and 0b0001)) {
    if (isTraceInProgress()) {
      traceEventStart(<>, %dirty, -1, <>)
    }
    val lambda = <block>{
      %composer.cache(%dirty and 0b1110 == 0b0100 || %dirty and 0b1000 != 0 && %composer.changedInstance(foo) or %dirty and 0b01110000 == 0b00100000 || %dirty and 0b01000000 != 0 && %composer.changedInstance(bar)) {
        {
          foo
          bar
        }
      }
    }
    if (isTraceInProgress()) {
      traceEventEnd()
    }
  } else {
    %composer.skipToGroupEnd()
  }
  %composer.endRestartGroup()?.updateScope { %composer: Composer?, %force: Int ->
    Test(foo, bar, %composer, updateChangedFlags(%changed or 0b0001))
  }
}
